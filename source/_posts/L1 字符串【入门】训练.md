#include <bits/stdc++.h>

	for(int i=0;s[i]!=0;i++)
	{
		if(s.substr(i,3)=="a:=")
		{
			if(s[i+3]=='a')a=a;
			else if(s[i+3]=='b')a=b;
			else if(s[i+3]=='c')a=c;
			else{a=s[i+3]-'0';}

		{
				if(s[i+3]=='a')b=a;

		}
